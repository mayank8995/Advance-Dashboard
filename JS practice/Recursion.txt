Find If Key Exists

function findIfKeyExists(obj, target){
    if( typeof obj !== 'object' || obj === null) return false;
    for(const key in obj){
        if(target === key) {
            return true;
        }
        const found = findIfKeyExists(obj[key],target);
        if(found) return true;
    }
    return false;
}

Given a JavaScript object where values can themselves be objects, return the maximum nesting depth.
const obj = {
  a: {},
  b: {
    c: {}
  },
  d: {
    e: {
      f: {}
    }
  }
}

function maxDepth(obj){
    let maxN = 0;
    if(Object.keys(obj).length === 0 || obj === null) return 0;
    for(const key in obj){
       maxN = Math.max(maxN,1 + maxDepth(obj[key]));
    }
    return maxN;
}
