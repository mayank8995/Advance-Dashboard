function maxDepth(obj){
    let maxN = 0;
    if(obj === null || Object.keys(obj).length === 0) return 0;
    for(const key in obj){
       maxN = Math.max(maxN,1 + maxDepth(obj[key]));
    }
    return maxN;
}

Flatten Nested Array (Frontend Relevant)

Input:

[1,[2,[3,4]],5]

Output:

[1,2,3,4,5]

Base Case: arr.length === 0 return []
What gets smaller?: arr to be traversed
Recursive Relationship: if(Array.isArray(arr[i])) flattenArray.push(...flatten(arr[i])) else flattenArray.push(arr[i]) return flattenArray
Traversal Type: array traversal
Return style: Child returns array
Parent merges arrays

function flattenArray(arr){
    if(arr.length === 0) return [];
    else{
      const b = [];
        for(let i=0;i<arr.length;i++){
            if(Array.isArray(arr[i])){
                b.push(...flattenArray(arr[i]))
            }else{
                b.push(arr[i])
            }
        }
      return b
    }
}

function nestedCommentSearch(obj,id,target){
  if(obj === null || typeof obj !== 'object') return null;
  else{
    for(const key in obj){
        if(key===id && obj[key] === target) {
           return obj;
