function findIfKeyExists(obj, target){
    if( typeof obj !== 'object' || obj === null) return false;
    for(const key in obj){
        if(target === key) {
            return true;
        }
        const found = findIfKeyExists(obj[key],target);
        if(found) return true;
    }
    return false;
}

function maxDepth(obj){
    let maxN = 0;
    if(obj === null || Object.keys(obj).length === 0) return 0;
    for(const key in obj){
       maxN = Math.max(maxN,1 + maxDepth(obj[key]));
    }
    return maxN;
}

function generateAllSubsequence(str, index, current){
        if(index === str.length) return [current]
         const included = generateAllSubsequence(str,index + 1,current + str[index]);  
         const excluded = generateAllSubsequence(str,index + 1, current);
         return [...included,...excluded]
}

function flattenArray(arr){
    if(arr.length === 0) return [];
    else{
      const b = [];
        for(let i=0;i<arr.length;i++){
            if(Array.isArray(arr[i])){
                b.push(...flattenArray(arr[i]))
            }else{
                b.push(arr[i])
            }
        }
      return b
    }
}

function findAllValuesForAKey(obj,target,values){
  if(obj === null || typeof obj !== 'object') return values;
  else{
    for(const key in obj){
        if(key === target) {
          values.push(obj[key]); 
        }
        findAllValuesForAKey(obj[key], target,values);
    }
  }
  return values;
}

function nestedCommentSearch(obj,id,target){
  if(obj === null || typeof obj !== 'object') return null;
  else{
    for(const key in obj){
        if(key===id && obj[key] === target) {
           return obj;
        }
       const searched = nestedCommentSearch(obj[key],id,target);
      if(searched) return searched
    }
  }
}
