function isPalindrome(str) {
  let i = 0; 
  let j = str.length - 1;
  while(i != j && i < j){
    if(str[i] !== str[j]) return false;
    i++;
    j--;
  }
  return true;
}
console.log(isPalindrome("malayalam"))

function findMax(arr) {
  let max = arr[0];
  let i = 1;
  while(i < arr.length){
    if(arr[i] > max){
          max = arr[i];
    }
    i++;
  }
  return max;
}

function charCount(str) {
  let obj = {};
  for(let i = 0; i < str.length; i++){
    let char = str[i]; 
    obj[char] = (obj[char] || 0) + 1;
  }
  return obj;
}

function moveZeroes(arr) {

    let fast = 0;

function longestSubstring(str) {

    let set = new Set();
    let slow = 0;
    let maxLength = 0;
    for(let fast = 0; fast < str.length; fast++){

      while(set.has(str[fast])){
        set.delete(str[slow]);
        slow++;
      }
      set.add(str[fast]);
      maxLength = Math.max(maxLength, fast - slow + 1);
    }
  return maxLength
}

function maxSumSubarray(arr,k){
    if(k > arr.length) return "No valid maxsum of subarray of size k exists"
    let slow = 0;
    let currentWindowSum = 0;
    let maxSum = Number.MIN_SAFE_INTEGER;
    for(let fast = 0; fast < arr.length; fast++){
        currentWindowSum += arr[fast];
        if(fast - slow + 1 === k){
            maxSum = Math.max(maxSum, currentWindowSum)
            currentWindowSum -= arr[slow];
            slow++;
        }
    }
    return maxSum;
}

function smallestContiguousSubarray(arr, target){
    let slow = 0;
    let currentwindowSum = 0;
    let smallestLength = Number.MAX_SAFE_INTEGER;
    let fast = 0;
  while( fast < arr.length){
    if(currentwindowSum < target){
      currentwindowSum += arr[fast];
    }
    if(currentwindowSum >= target){
        smallestLength = Math.min(smallestLength,fast-slow+1)
        currentwindowSum -= arr[slow];
        slow++;
      }
      if(currentwindowSum < target) fast++;
  }
  return smallestLength === Number.MAX_SAFE_INTEGER ? "No such subarray exists" : smallestLength
}
