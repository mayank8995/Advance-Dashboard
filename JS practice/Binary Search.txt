function binarySearch(arr, target){
    let low = 0;
    let high = arr.length - 1;
    let mid;
    while(low <= high){
        mid = low + Math.floor((high-low)/2)

        if(arr[mid] === target) return mid;

        if(arr[mid] > target){
             high = mid - 1;
        }else{
             low = mid + 1;
        }
    }

    return -1;
}

2.Problem: First Occurrence in a Sorted Array

Given a sorted array with duplicates:

[1, 2, 2, 2, 3, 4, 5]

Target:

2

Return index of the first Occurrence
function firstOccurrence(arr, x) {
    let low = 0;
    let high = arr.length - 1;
    let mid;
   let ans = -1;
    while (high >= low) {
        mid = low + Math.floor((high - low) / 2);


        if (arr[mid] == x){
            ans = mid;
            high = mid-1;
          continue;
        }
        if (arr[mid] > x)
            high = mid - 1;
        else {
            low  = mid + 1;
        }
            
    }
    return ans;
}

4.Problem: Search Insert Position

Given a sorted array of distinct integers, return:

The index if the target exists.
Otherwise, the index where it should be inserted to keep the array sorted.

Examples:

arr = [1,3,5,6], target = 5
→ 2
arr = [1,3,5,6], target = 2
→ 1
arr = [1,3,5,6], target = 7
→ 4
arr = [1,3,5,6], target = 0
→ 0

function searchInsertPosition(arr, x) {
    let low = 0;
    let high = arr.length - 1;
    let mid;

    while (high >= low) {
        mid = low + Math.floor((high - low) / 2);

        if (arr[mid] == x){
            return mid;
        }
        if (arr[mid] > x)
            high = mid - 1;
        else {
            low  = mid + 1;
        }
    }
    // at loop termination condition is low > high so no need to write low > high ? low : high
    return  low
}
