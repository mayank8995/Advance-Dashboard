function fileExplorerTraversal(obj,currentPath){
  if(obj === null || typeof obj !== 'object') return currentPath;
  else{
    for(const child of obj){
      let path = currentPath.length > 0 ? currentPath + '/' + child.name : currentPath + child.name
      currentPath = fileExplorerTraversal(child.children,path);
      const directory = fileExplorerTraversal(child.children,currentPath);
      if(directory) console.log(directory)
    }
  }
}
