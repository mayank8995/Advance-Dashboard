Generate All Valid Parentheses

Input:

n = 2

Output:

[
  "(())",
  "()()"
]

//shared accumulator method
function generateValidParanthesis(current, openCount, closeCount,n,arr){

    if(openCount === n && closeCount === n) return arr.push(current);
    else{
      if(openCount < n) {
         generateValidParanthesis(current + "(", openCount + 1, closeCount,n,arr);

      }
      if(closeCount < openCount){
          generateValidParanthesis(current + ")", openCount , closeCount + 1,n,arr);
     }
      }
    return arr
  }

//child passes to parent
function generateValidParanthesis(current, openCount, closeCount,n){

    if(openCount === n && closeCount === n) return [current];

      let open;
      let close;
      if(openCount < n) {
         open = generateValidParanthesis(current + "(", openCount + 1, closeCount,n);
      }else if(openCount === n){
        open = [];
      }
      if(closeCount < openCount){
          close = generateValidParanthesis(current + ")", openCount , closeCount + 1,n);
     }else if(closeCount === openCount){
       close = [];
     }
       
    return [...open, ...close]
  }
